import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.Statement;
import java.sql.ResultSet;
import java.sql.SQLException;

public class StudentRecords {

    // MySQL Database credentials
    static final String DB_URL = "jdbc:mysql://localhost:3306/school_db";
    static final String USER = "root";  // Replace with your MySQL username
    static final String PASS = "password";  // Replace with your MySQL password

    public static void main(String[] args) {
        Connection conn = null;
        Statement stmt = null;

        try {
            // 1. Establish a connection to the database
            System.out.println("Connecting to database...");
            conn = DriverManager.getConnection(DB_URL, USER, PASS);
            System.out.println("Connected to database successfully!");

            // 2. Create a statement object to execute SQL queries
            stmt = conn.createStatement();

            // 3. Create the "students" table if it doesn't exist
            String createTableSQL = "CREATE TABLE IF NOT EXISTS students (" +
                    "student_id INT AUTO_INCREMENT PRIMARY KEY, " +
                    "name VARCHAR(100), " +
                    "age INT, " +
                    "grade VARCHAR(10))";
            stmt.executeUpdate(createTableSQL);
            System.out.println("Table 'students' created successfully!");

            // 4. Insert student records
            String insertSQL1 = "INSERT INTO students (name, age, grade) VALUES ('John Doe', 18, 'A')";
            String insertSQL2 = "INSERT INTO students (name, age, grade) VALUES ('Jane Smith', 20, 'B')";
            String insertSQL3 = "INSERT INTO students (name, age, grade) VALUES ('Mark Davis', 19, 'A')";
            stmt.executeUpdate(insertSQL1);
            stmt.executeUpdate(insertSQL2);
            stmt.executeUpdate(insertSQL3);
            System.out.println("Records inserted successfully!");

            // 5. Retrieve and display all records from the "students" table
            String selectSQL = "SELECT * FROM students";
            ResultSet rs = stmt.executeQuery(selectSQL);

            // 6. Process and print the retrieved records
            System.out.println("Retrieving student records...");
            while (rs.next()) {
                int id = rs.getInt("student_id");
                String name = rs.getString("name");
                int age = rs.getInt("age");
                String grade = rs.getString("grade");
                System.out.printf("ID: %d, Name: %s, Age: %d, Grade: %s%n", id, name, age, grade);
            }

            // 7. Drop the "students" table after retrieving the records
            String dropTableSQL = "DROP TABLE IF EXISTS students";
            stmt.executeUpdate(dropTableSQL);
            System.out.println("Table 'students' dropped successfully!");

        } catch (SQLException e) {
            // Handle SQL exceptions
            e.printStackTrace();
        } finally {
            try {
                // 8. Clean up resources by closing the statement and connection
                if (stmt != null) stmt.close();
                if (conn != null) conn.close();
                System.out.println("Database connection closed.");
            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
}
