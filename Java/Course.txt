class Course {
    private String courseName;
    private String courseID;
    private String courseCategory;
    
    // Getter and Setter methods
    public String getCourseName() {
        return courseName;
    }
    
    public void setCourseName(String courseName) {
        this.courseName = courseName;
    }
    
    public String getCourseID() {
        return courseID;
    }
    
    public void setCourseID(String courseID) {
        this.courseID = courseID;
    }
    
    public String getCourseCategory() {
        return courseCategory;
    }
    
    public void setCourseCategory(String courseCategory) {
        this.courseCategory = courseCategory;
    }
    
    // Display course details
    public void printCourseDetails() {
        System.out.println("Course: " + courseName + ", ID: " + courseID + ", Category: " + courseCategory);
    }
}

public class CourseDemo {
    public static void main(String[] args) {
        Course course = new Course();
        course.setCourseName("Java Basics");
        course.setCourseID("101");
        course.setCourseCategory("Programming");
        
        course.printCourseDetails();
        
        // Update the course name
        course.setCourseName("Computing in Java");
        course.printCourseDetails();
    }
}
